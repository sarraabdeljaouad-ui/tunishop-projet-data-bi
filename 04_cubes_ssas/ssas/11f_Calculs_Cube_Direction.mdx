/* =====================================================================
   11f_Calculs_Cube_Direction.mdx — TuniShop — script de calcul du cube [Direction]
   ---------------------------------------------------------------------
   Cube 6 — Direction : synthèse des 5 domaines pour le tableau de bord de direction.
   12 KPI de pilotage (v1) recopiés À L'IDENTIQUE des scripts 11a à 11e (mêmes noms,
   mêmes formules) : un KPI affiché dans le cube Direction doit donner exactement
   la même valeur que dans son cube métier (contrôle : 12f_Recette_Cube_Direction.mdx).
   Déjà intégré dans Direction.cube (onglet « Calculs »). Ce fichier sert de référence.
   Groupes de mesures : Ventes, Commandes (cube Ventes) · Clients distincts, Marketing
   (cube Marketing Clients) · Stock mensuel (cube Stock) · Achats (cube Achats) ·
   Livraisons, Avis (cube Livraisons).
   Divisions protégées (NULL si dénominateur vide).

/* ---------- 1. Ventes (source : 11a) ---------- */
CREATE MEMBER CURRENTCUBE.[Measures].[CA Reconnu TTC] AS
    ([Measures].[Montant Total TTC], [Statut Commande].[Famille Statut].&[aboutie]),
    FORMAT_STRING = "#,##0.000", DISPLAY_FOLDER = 'KPI Direction\1 Ventes', VISIBLE = 1;
CREATE MEMBER CURRENTCUBE.[Measures].[Commandes Reconnues] AS
    ([Measures].[Nb Commandes], [Statut Commande].[Famille Statut].&[aboutie]),
    FORMAT_STRING = "#,##0", DISPLAY_FOLDER = 'KPI Direction\1 Ventes', VISIBLE = 1;
CREATE MEMBER CURRENTCUBE.[Measures].[CA HT Reconnu] AS
    ([Measures].[Montant Ligne HT], [Statut Commande].[Famille Statut].&[aboutie]),
    FORMAT_STRING = "#,##0.000", DISPLAY_FOLDER = 'KPI Direction\1 Ventes', VISIBLE = 1;
CREATE MEMBER CURRENTCUBE.[Measures].[Marge Brute Reconnue HT] AS
    ([Measures].[Marge Brute HT], [Statut Commande].[Famille Statut].&[aboutie]),
    FORMAT_STRING = "#,##0.000", DISPLAY_FOLDER = 'KPI Direction\1 Ventes', VISIBLE = 1;
CREATE MEMBER CURRENTCUBE.[Measures].[Taux Marge Brute] AS
    IIF([Measures].[CA HT Reconnu] = 0, NULL, [Measures].[Marge Brute Reconnue HT] / [Measures].[CA HT Reconnu]),
    FORMAT_STRING = "0.00%", DISPLAY_FOLDER = 'KPI Direction\1 Ventes', VISIBLE = 1;
CREATE MEMBER CURRENTCUBE.[Measures].[Panier Moyen TTC] AS
    IIF([Measures].[Commandes Reconnues] = 0, NULL, [Measures].[CA Reconnu TTC] / [Measures].[Commandes Reconnues]),
    FORMAT_STRING = "#,##0.00", DISPLAY_FOLDER = 'KPI Direction\1 Ventes', VISIBLE = 1;
CREATE MEMBER CURRENTCUBE.[Measures].[CA Reconnu A-1] AS
    ([Measures].[CA Reconnu TTC],
     ParallelPeriod([Date].[Calendrier].[Année], 1, [Date].[Calendrier].CurrentMember)),
    FORMAT_STRING = "#,##0.000", DISPLAY_FOLDER = 'KPI Direction\1 Ventes', VISIBLE = 1;
CREATE MEMBER CURRENTCUBE.[Measures].[Evolution CA Reconnu] AS
    IIF(IsEmpty([Measures].[CA Reconnu TTC]) OR IsEmpty([Measures].[CA Reconnu A-1]) OR [Measures].[CA Reconnu A-1] = 0, NULL,
        ([Measures].[CA Reconnu TTC] - [Measures].[CA Reconnu A-1]) / [Measures].[CA Reconnu A-1]),
    FORMAT_STRING = "0.0%", DISPLAY_FOLDER = 'KPI Direction\1 Ventes', VISIBLE = 1;

/* ---------- 2. Clients (source : 11e) ---------- */
-- Clients actifs = clients distincts, hors commandes en échec de paiement
CREATE MEMBER CURRENTCUBE.[Measures].[Clients Actifs] AS
    Aggregate(
        Except([Statut Commande].[Statut].[Statut].MEMBERS, { [Statut Commande].[Statut].&[paiement_echoue] }),
        [Measures].[Nb Clients]),
    FORMAT_STRING = "#,##0", DISPLAY_FOLDER = 'KPI Direction\2 Clients', VISIBLE = 1;

/* ---------- 3. Stock (source : 11b ; Valeur Stock HT = mesure de base LastNonEmpty) ---------- */
CREATE MEMBER CURRENTCUBE.[Measures].[Taux Rupture Stock] AS
    IIF([Measures].[Nb Inventaires] = 0, NULL, [Measures].[Nb Ruptures] / [Measures].[Nb Inventaires]),
    FORMAT_STRING = "0.00%", DISPLAY_FOLDER = 'KPI Direction\3 Stock', VISIBLE = 1;

/* ---------- 4. Achats (source : 11c ; Montant Achat HT = mesure de base) ---------- */
CREATE MEMBER CURRENTCUBE.[Measures].[Taux Service Fournisseur] AS

/* ---------- 5. Livraisons et satisfaction (source : 11d) ---------- */

/* ---------- 6. Marketing (source : 11e) ---------- */
CREATE MEMBER CURRENTCUBE.[Measures].[ROAS] AS
    IIF([Measures].[Depense HT] = 0, NULL, [Measures].[CA Attribue TTC] / [Measures].[Depense HT]),
    FORMAT_STRING = "0.000", DISPLAY_FOLDER = 'KPI Direction\6 Marketing', VISIBLE = 1;

/* =====================================================================
   v2 (29/10/2026) — comparaison à l'année précédente et feux de tendance
   A-1      : même KPI sur la période parallèle de l'année précédente (hiérarchie Calendrier).
   Evolution : variation relative (montants, nombres, panier, note, ROAS).
   Ecart     : différence en points de pourcentage (taux).
   Statut    : 1 = favorable, 0 = stable, -1 = défavorable, NULL sans année précédente.
               Stable : |évolution| < 2 % ou |écart| < 0,5 point. Pas de statut pour la valeur
               du stock et le montant des achats (une hausse n'est ni bonne ni mauvaise en soi).
   Couleur   : #1E7B4F (vert, favorable), #5B6B7C (gris, stable ou sans comparaison), #B42318 (rouge).
   Aucun objectif chiffré : les feux comparent à l'année précédente, pas à une cible.
   ===================================================================== */
