-- Requête 1 : les 12 KPI de direction par année (la ligne « All » = Total)
SELECT
    { [Measures].[CA Reconnu TTC],
        [Measures].[Evolution CA Reconnu],
        [Measures].[Taux Marge Brute],
        [Measures].[Panier Moyen TTC],
        [Measures].[Clients Actifs],
        [Measures].[Valeur Stock HT],
        [Measures].[Taux Rupture Stock],
        [Measures].[Montant Achat HT],
        [Measures].[Taux Service Fournisseur],
        [Measures].[OTD],
        [Measures].[Note Moyenne],
        [Measures].[ROAS] } ON COLUMNS,
    NON EMPTY { [Date].[Calendrier].[All], [Date].[Calendrier].[Année].MEMBERS } ON ROWS
FROM [Direction];

-- Requête 2 : preuve automatique « Direction = cubes métier ».
-- LookupCube relit la même mesure dans le cube d'origine, pour la même période.
-- Attendu : 0 dans TOUTES les cellules (ou un écart d'arrondi en 10^-6).
WITH
MEMBER [Measures].[Ecart 01 CA Reconnu TTC] AS
    [Measures].[CA Reconnu TTC]
    - LookupCube("Ventes", "([Measures].[CA Reconnu TTC], " + [Date].[Calendrier].CurrentMember.UniqueName + ")"),
    FORMAT_STRING = "#,##0.000000"
MEMBER [Measures].[Ecart 02 Taux Marge Brute] AS
    [Measures].[Taux Marge Brute]
    - LookupCube("Ventes", "([Measures].[Taux Marge Brute], " + [Date].[Calendrier].CurrentMember.UniqueName + ")"),
    FORMAT_STRING = "#,##0.000000"
MEMBER [Measures].[Ecart 03 Panier Moyen TTC] AS
    [Measures].[Panier Moyen TTC]
    - LookupCube("Ventes", "([Measures].[Panier Moyen TTC], " + [Date].[Calendrier].CurrentMember.UniqueName + ")"),
    FORMAT_STRING = "#,##0.000000"
MEMBER [Measures].[Ecart 04 Clients Actifs] AS
    [Measures].[Clients Actifs]
    - LookupCube("Marketing Clients", "([Measures].[Clients Actifs], " + [Date].[Calendrier].CurrentMember.UniqueName + ")"),
    FORMAT_STRING = "#,##0.000000"
MEMBER [Measures].[Ecart 05 Valeur Stock HT] AS
    [Measures].[Valeur Stock HT]
    - LookupCube("Stock", "([Measures].[Valeur Stock HT], " + [Date].[Calendrier].CurrentMember.UniqueName + ")"),
    FORMAT_STRING = "#,##0.000000"
MEMBER [Measures].[Ecart 06 Taux Rupture Stock] AS
    [Measures].[Taux Rupture Stock]
    - LookupCube("Stock", "([Measures].[Taux Rupture Stock], " + [Date].[Calendrier].CurrentMember.UniqueName + ")"),
    FORMAT_STRING = "#,##0.000000"
MEMBER [Measures].[Ecart 07 Montant Achat HT] AS
    [Measures].[Montant Achat HT]
    - LookupCube("Achats", "([Measures].[Montant Achat HT], " + [Date].[Calendrier].CurrentMember.UniqueName + ")"),
    FORMAT_STRING = "#,##0.000000"
MEMBER [Measures].[Ecart 08 Taux Service Fournisseur] AS
    [Measures].[Taux Service Fournisseur]
    - LookupCube("Achats", "([Measures].[Taux Service Fournisseur], " + [Date].[Calendrier].CurrentMember.UniqueName + ")"),
    FORMAT_STRING = "#,##0.000000"
MEMBER [Measures].[Ecart 09 OTD] AS
    [Measures].[OTD]
    - LookupCube("Livraisons", "([Measures].[OTD], " + [Date].[Calendrier].CurrentMember.UniqueName + ")"),
    FORMAT_STRING = "#,##0.000000"
MEMBER [Measures].[Ecart 10 Note Moyenne] AS
    [Measures].[Note Moyenne]
    - LookupCube("Livraisons", "([Measures].[Note Moyenne], " + [Date].[Calendrier].CurrentMember.UniqueName + ")"),
    FORMAT_STRING = "#,##0.000000"
MEMBER [Measures].[Ecart 11 ROAS] AS
    [Measures].[ROAS]
    - LookupCube("Marketing Clients", "([Measures].[ROAS], " + [Date].[Calendrier].CurrentMember.UniqueName + ")"),
    FORMAT_STRING = "#,##0.000000"
SELECT
    { [Measures].[Ecart 01 CA Reconnu TTC],
        [Measures].[Ecart 02 Taux Marge Brute],
        [Measures].[Ecart 03 Panier Moyen TTC],
        [Measures].[Ecart 04 Clients Actifs],
        [Measures].[Ecart 05 Valeur Stock HT],
        [Measures].[Ecart 06 Taux Rupture Stock],
        [Measures].[Ecart 07 Montant Achat HT],
        [Measures].[Ecart 08 Taux Service Fournisseur],
        [Measures].[Ecart 09 OTD],
        [Measures].[Ecart 10 Note Moyenne],
        [Measures].[Ecart 11 ROAS] } ON COLUMNS,
    { [Date].[Calendrier].[All], [Date].[Calendrier].[Année].MEMBERS } ON ROWS
FROM [Direction];

-- Requête 3 : comportement des dimensions non liées (IgnoreUnrelatedDimensions = vrai, valeur par défaut)
-- Filtre sur le canal « application_mobile » : CA Reconnu et Panier Moyen changent ;
-- Valeur Stock HT, Montant Achat HT, OTD et ROAS restent au total (ces groupes ne sont pas liés au Canal).
SELECT
    { [Measures].[CA Reconnu TTC], [Measures].[Panier Moyen TTC],
      [Measures].[Valeur Stock HT], [Measures].[Montant Achat HT], [Measures].[OTD], [Measures].[ROAS] } ON COLUMNS,
    { [Canal].[Canal].[All], [Canal].[Canal].&[application_mobile] } ON ROWS
FROM [Direction]
WHERE [Date].[Calendrier].[Année].&[2025];

-- Requête 4 (v2, 29/10/2026) : comparaison à l'année précédente et feux de tendance
-- Attendu (valeurs du cube) :            2024                  2025
--   Evolution CA Reconnu              +48,4 %   statut 1    +42,1 %   statut 1
--   Ecart Taux Marge Brute           +0,51 pt   statut 1   -0,00 pt   statut 0   (2024 : juste au-dessus du seuil de 0,5 pt)
--   Evolution Panier Moyen             +4,7 %   statut 1     +1,1 %   statut 0
--   Evolution Clients Actifs          +37,5 %   statut 1    +33,4 %   statut 1
--   Evolution Valeur Stock            +22,7 %   (pas de statut)  +26,7 %
--   Ecart Taux Rupture Stock         +0,31 pt   statut 0   +0,51 pt   statut -1  (2025 : juste au-dessus du seuil)
--   Evolution Montant Achat           +57,0 %   (pas de statut)  +42,1 %
--   Ecart Taux Service Fournisseur   -0,21 pt   statut 0   +0,56 pt   statut 1
--   Ecart OTD                        -4,50 pt   statut -1  -3,26 pt   statut -1
--   Evolution Note Moyenne             -1,0 %   statut 0     -0,4 %   statut 0
--   Evolution ROAS                     +8,8 %   statut 1    +11,0 %   statut 1
-- Ligne 2023 : A-1, évolutions et statuts vides (pas d'année 2022).
-- Contrôle des A-1 : chaque « X A-1 » de 2025 doit égaler « X » de 2024 (requête 1).
SELECT
    { [Measures].[Evolution CA Reconnu], [Measures].[Statut CA Reconnu],
      [Measures].[Ecart Taux Marge Brute], [Measures].[Statut Taux Marge Brute],
      [Measures].[Evolution Panier Moyen], [Measures].[Statut Panier Moyen],
      [Measures].[Evolution Clients Actifs], [Measures].[Statut Clients Actifs],
      [Measures].[Evolution Valeur Stock],
      [Measures].[Ecart Taux Rupture Stock], [Measures].[Statut Taux Rupture Stock],
      [Measures].[Evolution Montant Achat],
      [Measures].[Ecart Taux Service Fournisseur], [Measures].[Statut Taux Service Fournisseur],
      [Measures].[Ecart OTD], [Measures].[Statut OTD],
      [Measures].[Evolution Note Moyenne], [Measures].[Statut Note Moyenne],
      [Measures].[Evolution ROAS], [Measures].[Statut ROAS] } ON COLUMNS,
    NON EMPTY [Date].[Calendrier].[Année].MEMBERS ON ROWS
FROM [Direction];
