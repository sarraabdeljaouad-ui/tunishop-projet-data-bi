-- Requête 1 : les KPI du cube par année (la ligne « All » = Total)
SELECT
    { [Measures].[Depense HT],
        [Measures].[CPC],
        [Measures].[CPM],
        [Measures].[Taux Clic],
        [Measures].[Taux Conversion],
        [Measures].[CAC],
        [Measures].[ROAS],
        [Measures].[Nouveaux Clients],
        [Measures].[Clients Actifs],
        [Measures].[Taux Reachat] } ON COLUMNS,
    NON EMPTY { [Date].[Calendrier].[All], [Date].[Calendrier].[Année].MEMBERS } ON ROWS
FROM [Marketing Clients];

-- Requête 2 : ROAS par canal média
-- Attendu : emailing 5,874 · facebook_ads 3,339 · sms 3,195 · google_ads 3,073 · instagram_ads 2,947 · influence 2,091 · tiktok_ads 1,614
SELECT
    { [Measures].[Depense HT], [Measures].[CA Attribue TTC], [Measures].[ROAS] } ON COLUMNS,
    NON EMPTY [Campagne].[Canal Média].[Canal Média].MEMBERS ON ROWS
FROM [Marketing Clients];
