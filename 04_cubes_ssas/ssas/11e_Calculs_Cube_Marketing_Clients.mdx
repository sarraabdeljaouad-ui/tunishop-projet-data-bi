/* =====================================================================
   11e_Calculs_Cube_Marketing_Clients.mdx — TuniShop — script de calcul du cube [Marketing Clients]
   ---------------------------------------------------------------------
   Cube 5 — Marketing et clients : campagnes, acquisition, fidélisation.
   Où le coller : Visual Studio → cube Marketing Clients → onglet « Calculs »
   → « Affichage du script » → Ctrl+A → coller → Ctrl+S → Déployer.
   Mesures de base attendues dans ce cube : groupes Marketing (Depense HT, Impressions, Clics, Commandes Attribuees, Nouveaux Clients Attribues, CA Attribue TTC), Commandes (Nb Commandes, Montant Total TTC, Nb Premieres Commandes) et Clients distincts (Nb Clients)
   Règles : Dictionnaire_KPI.xlsx (partie 4). Divisions protégées (NULL si dénominateur vide).
   ===================================================================== */
CALCULATE;

CREATE MEMBER CURRENTCUBE.[Measures].[CPC] AS
    IIF([Measures].[Clics] = 0, NULL, [Measures].[Depense HT] / [Measures].[Clics]),
    FORMAT_STRING = "0.000", DISPLAY_FOLDER = 'KPI\Marketing', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[CPM] AS
    IIF([Measures].[Impressions] = 0, NULL, [Measures].[Depense HT] * 1000 / [Measures].[Impressions]),
    FORMAT_STRING = "0.000", DISPLAY_FOLDER = 'KPI\Marketing', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Taux Clic] AS
    IIF([Measures].[Impressions] = 0, NULL, [Measures].[Clics] / [Measures].[Impressions]),
    FORMAT_STRING = "0.000%", DISPLAY_FOLDER = 'KPI\Marketing', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Taux Conversion] AS
    IIF([Measures].[Clics] = 0, NULL, [Measures].[Commandes Attribuees] / [Measures].[Clics]),
    FORMAT_STRING = "0.000%", DISPLAY_FOLDER = 'KPI\Marketing', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[CAC] AS
    IIF([Measures].[Nouveaux Clients Attribues] = 0, NULL, [Measures].[Depense HT] / [Measures].[Nouveaux Clients Attribues]),
    FORMAT_STRING = "#,##0.00", DISPLAY_FOLDER = 'KPI\Marketing', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[ROAS] AS
    IIF([Measures].[Depense HT] = 0, NULL, [Measures].[CA Attribue TTC] / [Measures].[Depense HT]),
    FORMAT_STRING = "0.000", DISPLAY_FOLDER = 'KPI\Marketing', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[CA Reconnu TTC] AS
    ([Measures].[Montant Total TTC], [Statut Commande].[Famille Statut].&[aboutie]),
    FORMAT_STRING = "#,##0.000", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

-- Clients actifs = clients distincts, hors commandes en échec de paiement
CREATE MEMBER CURRENTCUBE.[Measures].[Clients Actifs] AS
    Aggregate(
        Except([Statut Commande].[Statut].[Statut].MEMBERS, { [Statut Commande].[Statut].&[paiement_echoue] }),
        [Measures].[Nb Clients]),
    FORMAT_STRING = "#,##0", DISPLAY_FOLDER = 'KPI\Clients', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Nouveaux Clients] AS
    [Measures].[Nb Premieres Commandes],
    FORMAT_STRING = "#,##0", DISPLAY_FOLDER = 'KPI\Clients', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[CA Par Client Actif] AS
    IIF([Measures].[Clients Actifs] = 0, NULL, [Measures].[CA Reconnu TTC] / [Measures].[Clients Actifs]),
    FORMAT_STRING = "#,##0.00", DISPLAY_FOLDER = 'KPI\Clients', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Commandes Hors Echec Paiement] AS
    Aggregate(
        Except([Statut Commande].[Statut].[Statut].MEMBERS, { [Statut Commande].[Statut].&[paiement_echoue] }),
        [Measures].[Nb Commandes]),
    FORMAT_STRING = "#,##0", DISPLAY_FOLDER = 'KPI\Clients', VISIBLE = 0;

-- Clients ayant passé au moins 2 commandes (hors échec de paiement) sur la période affichée
CREATE MEMBER CURRENTCUBE.[Measures].[Clients Fideles] AS
    Count(Filter(EXISTING [Client].[Client].[Client].MEMBERS,
                 [Measures].[Commandes Hors Echec Paiement] >= 2)),
    FORMAT_STRING = "#,##0", DISPLAY_FOLDER = 'KPI\Clients', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Taux Reachat] AS
    IIF([Measures].[Clients Actifs] = 0, NULL, [Measures].[Clients Fideles] / [Measures].[Clients Actifs]),
    FORMAT_STRING = "0.00%", DISPLAY_FOLDER = 'KPI\Clients', VISIBLE = 1;

/* =====================================================================
   v2 (29/10/2026) — comparaison à l'année précédente et feux de tendance (même règle que le cube Direction)
   A-1 : même KPI sur la période parallèle de l'année précédente (hiérarchie Calendrier).
   Evolution : variation relative ; Ecart : différence en points de pourcentage (taux).
   Statut : 1 favorable, 0 stable (|évolution| < 2 % ou |écart| < 0,5 point), -1 défavorable ; NULL sans année précédente.
   Couleur : #1E7B4F, #5B6B7C, #B42318. Pas de feu pour les KPI sans sens favorable (volumes, parts, remises, valeur du stock).
   ===================================================================== */
