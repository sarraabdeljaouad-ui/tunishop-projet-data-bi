-- Requête 1 : les KPI du cube par année (la ligne « All » = Total)
SELECT
    { [Measures].[CA Commande TTC],
        [Measures].[Commandes Passees],
        [Measures].[CA Reconnu TTC],
        [Measures].[Commandes Reconnues],
        [Measures].[Rembourse TTC],
        [Measures].[CA Net TTC],
        [Measures].[CA HT Reconnu],
        [Measures].[Marge Brute Reconnue HT],
        [Measures].[Taux Marge Brute],
        [Measures].[Panier Moyen TTC],
        [Measures].[Articles Par Commande],
        [Measures].[Taux Remise Produit],
        [Measures].[Part Application Mobile],
        [Measures].[Part Paiement A La Livraison],
        [Measures].[Taux Annulation],
        [Measures].[Taux Echec Paiement],
        [Measures].[Taux Refus Colis] } ON COLUMNS,
    NON EMPTY { [Date].[Calendrier].[All], [Date].[Calendrier].[Année].MEMBERS } ON ROWS
FROM [Ventes];

-- Requête 2 : CA reconnu par catégorie en 2025
SELECT
    { [Measures].[CA HT Reconnu], [Measures].[Marge Brute Reconnue HT], [Measures].[Taux Marge Brute] } ON COLUMNS,
    NON EMPTY [Produit].[Catégorie].[Catégorie].MEMBERS ON ROWS
FROM [Ventes]
WHERE [Date].[Calendrier].[Année].&[2025];

-- Requête 3 : évolution sur un an — attendu : 2024 = +48,4 % · 2025 = +42,1 %
SELECT
    { [Measures].[CA Reconnu TTC], [Measures].[CA Reconnu A-1], [Measures].[Evolution CA Reconnu] } ON COLUMNS,
    NON EMPTY [Date].[Calendrier].[Année].MEMBERS ON ROWS
FROM [Ventes];
