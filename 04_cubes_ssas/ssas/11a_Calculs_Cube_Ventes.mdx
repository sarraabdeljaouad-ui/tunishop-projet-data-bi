/* =====================================================================
   11a_Calculs_Cube_Ventes.mdx — TuniShop — script de calcul du cube [Ventes]
   ---------------------------------------------------------------------
   Cube 1 — Ventes : chiffre d'affaires, marge, panier, canaux, annulations, paiements.
   Où le coller : Visual Studio → cube Ventes → onglet « Calculs »
   → « Affichage du script » → Ctrl+A → coller → Ctrl+S → Déployer.
   Mesures de base attendues dans ce cube : groupes Ventes, Commandes, Paiements (Nb Lignes Vente, Montant Total TTC, Nb Commandes, Montant Ligne HT, Marge Brute HT, Montant Brut TTC, Remise Promo TTC, Nb Articles, Montant Rembourse Commande TTC, Nb Tentatives Paiement, Nb Paiements Reussis…)

CALCULATE;

CREATE MEMBER CURRENTCUBE.[Measures].[CA Commande TTC] AS
    [Measures].[Montant Total TTC],
    FORMAT_STRING = "#,##0.000", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Commandes Passees] AS
    [Measures].[Nb Commandes],
    FORMAT_STRING = "#,##0", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[CA Reconnu TTC] AS
    ([Measures].[Montant Total TTC], [Statut Commande].[Famille Statut].&[aboutie]),
    FORMAT_STRING = "#,##0.000", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Commandes Reconnues] AS
    ([Measures].[Nb Commandes], [Statut Commande].[Famille Statut].&[aboutie]),
    FORMAT_STRING = "#,##0", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Rembourse TTC] AS
    ([Measures].[Montant Rembourse Commande TTC], [Statut Commande].[Famille Statut].&[aboutie]),
    FORMAT_STRING = "#,##0.000", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[CA Net TTC] AS
    [Measures].[CA Reconnu TTC] - [Measures].[Rembourse TTC],
    FORMAT_STRING = "#,##0.000", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[CA HT Reconnu] AS
    ([Measures].[Montant Ligne HT], [Statut Commande].[Famille Statut].&[aboutie]),
    FORMAT_STRING = "#,##0.000", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Marge Brute Reconnue HT] AS
    ([Measures].[Marge Brute HT], [Statut Commande].[Famille Statut].&[aboutie]),
    FORMAT_STRING = "#,##0.000", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Taux Marge Brute] AS
    IIF([Measures].[CA HT Reconnu] = 0, NULL, [Measures].[Marge Brute Reconnue HT] / [Measures].[CA HT Reconnu]),
    FORMAT_STRING = "0.00%", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Panier Moyen TTC] AS
    IIF([Measures].[Commandes Reconnues] = 0, NULL, [Measures].[CA Reconnu TTC] / [Measures].[Commandes Reconnues]),
    FORMAT_STRING = "#,##0.00", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Articles Par Commande] AS
    IIF([Measures].[Commandes Reconnues] = 0, NULL,
        ([Measures].[Nb Articles], [Statut Commande].[Famille Statut].&[aboutie]) / [Measures].[Commandes Reconnues]),
    FORMAT_STRING = "0.000", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Taux Remise Produit] AS
    IIF(([Measures].[Montant Brut TTC], [Statut Commande].[Famille Statut].&[aboutie]) = 0, NULL,
        ([Measures].[Remise Promo TTC], [Statut Commande].[Famille Statut].&[aboutie])
        / ([Measures].[Montant Brut TTC], [Statut Commande].[Famille Statut].&[aboutie])),
    FORMAT_STRING = "0.00%", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

-- Parts : numérateur = un canal / un mode ; dénominateur = tous les canaux / tous les modes
CREATE MEMBER CURRENTCUBE.[Measures].[Part Application Mobile] AS
    IIF(([Measures].[Nb Commandes], [Canal].[Canal].[All]) = 0, NULL,
        ([Measures].[Nb Commandes], [Canal].[Canal].&[application_mobile])
        / ([Measures].[Nb Commandes], [Canal].[Canal].[All])),
    FORMAT_STRING = "0.00%", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Part Paiement A La Livraison] AS
    IIF(([Measures].[Nb Commandes], [Mode Paiement].[Mode Paiement].[All]) = 0, NULL,
        ([Measures].[Nb Commandes], [Mode Paiement].[Mode Paiement].&[paiement_a_la_livraison])
        / ([Measures].[Nb Commandes], [Mode Paiement].[Mode Paiement].[All])),
    FORMAT_STRING = "0.00%", DISPLAY_FOLDER = 'KPI\Ventes', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Taux Annulation] AS
    IIF(([Measures].[Nb Commandes], Root([Statut Commande])) = 0, NULL,
        ([Measures].[Nb Commandes], [Statut Commande].[Statut].&[annulee])
        / ([Measures].[Nb Commandes], Root([Statut Commande]))),
    FORMAT_STRING = "0.00%", DISPLAY_FOLDER = 'KPI\Operations', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Taux Refus Colis] AS
    IIF(([Measures].[Nb Commandes], [Statut Commande].[Famille Statut].&[aboutie])
        + ([Measures].[Nb Commandes], [Statut Commande].[Statut].&[livraison_echouee]) = 0, NULL,
        ([Measures].[Nb Commandes], [Statut Commande].[Statut].&[livraison_echouee])
        / (([Measures].[Nb Commandes], [Statut Commande].[Famille Statut].&[aboutie])
           + ([Measures].[Nb Commandes], [Statut Commande].[Statut].&[livraison_echouee]))),
    FORMAT_STRING = "0.00%", DISPLAY_FOLDER = 'KPI\Operations', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[Taux Echec Paiement] AS
    IIF([Measures].[Nb Tentatives Paiement] = 0, NULL,
        1 - [Measures].[Nb Paiements Reussis] / [Measures].[Nb Tentatives Paiement]),
    FORMAT_STRING = "0.00%", DISPLAY_FOLDER = 'KPI\Operations', VISIBLE = 1;

CREATE MEMBER CURRENTCUBE.[Measures].[CA Reconnu A-1] AS

CREATE MEMBER CURRENTCUBE.[Measures].[Evolution CA Reconnu] AS
