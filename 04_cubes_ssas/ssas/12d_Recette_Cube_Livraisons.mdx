-- Requête 1 : les KPI du cube par année (la ligne « All » = Total)
SELECT
    { [Measures].[Delai Moyen Livraison J],
        [Measures].[OTD],
        [Measures].[Taux Retard],
        [Measures].[Retard Moyen J],
        [Measures].[Cout Transport Moyen HT],
        [Measures].[Taux Livraison 1er Passage],
        [Measures].[Taux Retours],
        [Measures].[Delai Moyen Remboursement J],
        [Measures].[Note Moyenne],
        [Measures].[Part Avis Negatifs] } ON COLUMNS,
    NON EMPTY { [Date].[Calendrier].[All], [Date].[Calendrier].[Année].MEMBERS } ON ROWS
FROM [Livraisons];

-- Requête 2 : taux de retard par transporteur et par semestre (scénario SahelLog)
-- Attendu SahelLog : 2023-S1 7,72 % · 2023-S2 5,60 % · 2024-S1 8,52 % · 2024-S2 37,17 % · 2025-S1 38,20 % · 2025-S2 36,45 %
-- Attendu sur toute la période : Poste Colis Plus 18,03 % · Rapid Colis 6,78 % · SahelLog 26,03 % · TunExpress 5,09 %
SELECT
    NON EMPTY [Date].[Calendrier].[Semestre].MEMBERS ON COLUMNS,
    NON EMPTY [Transporteur].[Transporteur].[Transporteur].MEMBERS ON ROWS
FROM [Livraisons]
WHERE [Measures].[Taux Retard];
