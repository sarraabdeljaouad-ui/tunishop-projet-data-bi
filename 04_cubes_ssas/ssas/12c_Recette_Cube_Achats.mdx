-- Requête 1 : les KPI du cube par année (la ligne « All » = Total)
SELECT
    { [Measures].[Montant Achat HT],
        [Measures].[Taux Rupture Fournisseur],
        [Measures].[Taux Service Fournisseur],
        [Measures].[Delai Moyen Appro J],
        [Measures].[Retard Moyen Appro J] } ON COLUMNS,
    NON EMPTY { [Date].[Calendrier].[All], [Date].[Calendrier].[Année].MEMBERS } ON ROWS
FROM [Achats];

-- Requête 2 : le fournisseur FRN025 (Shenzhen Bright Tech) au 2e trimestre 2024
-- Attendu : retard moyen d'environ 34 jours sur les bons émis au T2 2024 (scénario du jeu de données)
SELECT
    { [Measures].[Montant Achat HT], [Measures].[Retard Moyen Appro J], [Measures].[Taux Rupture Fournisseur] } ON COLUMNS,
    NON EMPTY [Date].[Calendrier].[Trimestre].MEMBERS ON ROWS
FROM [Achats]
WHERE [Fournisseur].[Fournisseur].[Shenzhen Bright Tech];
