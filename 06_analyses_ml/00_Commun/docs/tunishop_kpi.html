tunishop_kpi.py — Calcul de référence des KPI TuniShop (phase BI / analyse)

Un seul endroit où chaque KPI est défini et calculé en Python. Les notebooks d'analyse (EDA, segmentation, classification, association) importent ce module : les chiffres sont donc identiques d'un bout à l'autre du projet, et identiques à ceux de Power BI / Tableau (recette).

Source des données : les vues bi.v_* de TuniShop_DWH (jamais les tables dw.*).

Utilisation type (dans un notebook) : import tunishop_kpi as tk cn = tk.connecter(serveur="localhost") # authentification Windows vues = tk.charger_vues(cn) # 23 vues -> DataFrames p = tk.preparer(vues) # tables enrichies resultats = tk.calculer_kpi(p) # tous les KPI (global + années + détails) controles = tk.controler_coherence(p) # tendances attendues tk.exporter_excel(resultats, controles, "kpi_reference.xlsx")

 46def connecter(serveur: str = "localhost", base: str = "TuniShop_DWH", auth: str = "windows",
 47              utilisateur: str | None = None, pilote: str = "ODBC Driver 17 for SQL Server"):
 48    """Ouvre une connexion pyodbc vers le DWH.
 49    auth = "windows" (Trusted_Connection) ou "sql" (mot de passe lu dans la
 50    variable d'environnement TUNISHOP_MDP : jamais de mot de passe dans le code)."""
 51    import pyodbc
 52    cs = f"DRIVER={{{pilote}}};SERVER={serveur};DATABASE={base};"
 53    if auth == "windows":
 54        cs += "Trusted_Connection=yes;"
 55    else:
 56        mdp = os.environ.get("TUNISHOP_MDP")
 57        if not utilisateur or not mdp:
 58            raise ValueError("Authentification SQL : renseigner utilisateur et la variable TUNISHOP_MDP.")
 59        cs += f"UID={utilisateur};PWD={mdp};"
 60    return pyodbc.connect(cs)

 63def _convertir_types(df: pd.DataFrame) -> pd.DataFrame:
 64    """pyodbc renvoie les DECIMAL en objets Decimal : on les convertit en float."""
 65    for col in df.columns:
 66        if df[col].dtype == object:
 67            echantillon = df[col].dropna().head(50)
 68            if len(echantillon) and all(isinstance(v, Decimal) for v in echantillon):
 69                df[col] = df[col].astype(float)
 70    return df

 73def charger_vues(cn, vues: list[str] = VUES) -> dict[str, pd.DataFrame]:
 74    """Lit chaque vue bi.v_* dans un DataFrame (clé = nom de la vue sans 'v_')."""
 75    resultat = {}
 76    with warnings.catch_warnings():
 77        warnings.simplefilter("ignore", UserWarning)   # avertissement pandas « SQLAlchemy »
 78        for v in vues:
 79            df = pd.read_sql(f"SELECT * FROM bi.[{v}]", cn)
 80            resultat[v[2:]] = _convertir_types(df)
 81            print(f"  {v:<30} {len(df):>8,} lignes".replace(",", " "))
 82    return resultat

124def preparer(v: dict[str, pd.DataFrame]) -> dict[str, pd.DataFrame]:
125    """Joint les faits à leurs dimensions et ajoute les colonnes d'analyse
126    (annee, famille de statut, catégorie, transporteur...)."""
127    d = {k: x.copy() for k, x in v.items()}
128    statut = d["dim_statut_commande"][["statut_commande_sk", "statut_commande", "famille_statut"]]
129    canal = d["dim_canal"][["canal_sk", "canal", "appareil"]]
130    mode = d["dim_mode_paiement"][["mode_paiement_sk", "mode_paiement"]]
131    client = d["dim_client"][["client_sk", "client_id", "segment_fidelite", "type_client"]]
132    geo = d["dim_geographie"][["geographie_sk", "region", "gouvernorat"]]
133    produit = d["dim_produit"][["produit_sk", "produit_id", "categorie", "sous_categorie", "fournisseur_id"]]
134    transp = d["dim_transporteur"][["transporteur_sk", "nom_transporteur"]].rename(columns={"nom_transporteur": "transporteur"})
135    campagne = d["dim_campagne"][["campagne_sk", "canal_media"]]
136    motif = d["dim_motif_retour"][["motif_retour_sk", "type_remboursement", "motif"]]
137    fourn = d["dim_fournisseur"][["fournisseur_sk", "fournisseur_id", "nom_fournisseur"]]
138
139    # --- Commandes (grain : commande)
140    cmd = (d["fact_commandes"].merge(statut, on="statut_commande_sk", how="left")
141           .merge(canal, on="canal_sk", how="left").merge(mode, on="mode_paiement_sk", how="left")
142           .merge(client, on="client_sk", how="left").merge(geo, on="geographie_sk", how="left"))
143    cmd["annee"] = _annee(cmd["date_commande_sk"])
144    cmd["date"] = _date(cmd["date_commande_sk"])
145    cmd["est_aboutie"] = cmd["famille_statut"].eq(FAMILLE_ABOUTIE)
146    cmd["est_premiere_commande"] = _bool(cmd["est_premiere_commande"])
147
148    # --- Remboursements (grain : demande de remboursement)
149    rmb = (d["fact_remboursements"].merge(motif, on="motif_retour_sk", how="left")
150           .merge(produit[["produit_sk", "produit_id", "categorie"]], on="produit_sk", how="left"))
151    rmb["annee"] = _annee(rmb["date_demande_sk"])
152    rmb["est_rembourse"] = rmb["statut"].eq("rembourse")
153
154    # Lignes retournées : retour accepté ou en cours (statut <> refuse), relié par commande + produit
155    retours = rmb[rmb["type_remboursement"].eq("retour") & rmb["statut"].ne("refuse") & rmb["produit_id"].notna()]
156    cles_retour = set(zip(retours["commande_id"], retours["produit_id"]))
157
158    # --- Lignes de commande (grain : ligne)
159    lig = (d["fact_ventes"].merge(statut, on="statut_commande_sk", how="left")
160           .merge(produit, on="produit_sk", how="left"))
161    lig["annee"] = _annee(lig["date_commande_sk"])
162    lig["est_aboutie"] = lig["famille_statut"].eq(FAMILLE_ABOUTIE)
163    lig["est_retournee"] = [(c, p) in cles_retour for c, p in zip(lig["commande_id"], lig["produit_id"])]
164
165    # --- Livraisons (grain : livraison)
166    liv = (d["fact_livraisons"].merge(transp, on="transporteur_sk", how="left")
167           .merge(geo, on="geographie_sk", how="left"))
168    liv["annee"] = _annee(liv["date_expedition_sk"])
169    liv["semestre"] = np.where(liv["date_expedition_sk"] > 0,
170                               (liv["date_expedition_sk"] // 10000).astype(str) + "-S"
171                               + np.where((liv["date_expedition_sk"] // 100) % 100 <= 6, "1", "2"), None)
172    for c in ["est_livree", "est_livree_a_temps", "est_premiere_tentative"]:
173        liv[c] = _bool(liv[c])
174
175    # --- Avis (grain : avis) + retard de la livraison de la commande
176    retard_cmd = liv[liv["est_livree"]].groupby("commande_id")["retard_j"].max()
177    avis = d["fact_avis"].merge(produit[["produit_sk", "categorie"]], on="produit_sk", how="left")
178    avis["annee"] = _annee(avis["date_avis_sk"])
179    r = avis["commande_id"].map(retard_cmd)
180    avis["tranche_retard"] = np.select([r.isna(), r.eq(0), r.le(2)], ["non livrée", "0 j", "1-2 j"], "3 j et +")
181
182    # --- Marketing (grain : campagne x jour)
183    mkt = d["fact_marketing_quotidien"].merge(campagne, on="campagne_sk", how="left")
184    mkt["annee"] = _annee(mkt["date_sk"])
185
186    # --- Stock mensuel (snapshot) et mouvements
187    stk = d["fact_stock_mensuel"].merge(produit[["produit_sk", "categorie"]], on="produit_sk", how="left")
188    stk["annee"] = _annee(stk["date_sk"])
189    stk["est_rupture"] = _bool(stk["est_rupture"])
190    mvt = d["fact_mouvements_stock"].merge(produit[["produit_sk", "categorie"]], on="produit_sk", how="left")
191    mvt["annee"] = _annee(mvt["date_sk"])
192    mvt["date"] = _date(mvt["date_sk"])
193
194    # --- Achats fournisseurs (grain : ligne de bon de commande)
195    ach = d["fact_achats_fournisseur"].merge(fourn, on="fournisseur_sk", how="left")
196    ach["annee"] = _annee(ach["date_commande_sk"])
197    ach["trimestre"] = np.where(ach["date_commande_sk"] > 0,
198                                (ach["date_commande_sk"] // 10000).astype(str) + "-T"
199                                + (((ach["date_commande_sk"] // 100) % 100 - 1) // 3 + 1).astype(str), None)
200
201    # --- Paiements (grain : tentative)
202    pai = d["fact_paiements"].merge(mode, on="mode_paiement_sk", how="left")
203    pai["annee"] = _annee(pai["date_paiement_sk"])
204    pai["est_reussi"] = _bool(pai["est_reussi"])
205
206    # --- Clients (version courante, hors membres techniques)
207    cli = d["dim_client"]
208    cli = cli[(cli["client_sk"] > 0) & _bool(cli["est_courant"]) & ~_bool(cli["est_supprime"])].copy()
209
210    return {"cmd": cmd, "lig": lig, "liv": liv, "rmb": rmb, "avis": avis, "mkt": mkt,
211            "stk": stk, "mvt": mvt, "ach": ach, "pai": pai, "cli": cli}

244def _couverture(d):
245    stk, mvt = d["stk"], d["mvt"]
246    if stk.empty:
247        return np.nan
248    dern = stk["date_sk"].max()
249    stock_qte = stk.loc[stk["date_sk"].eq(dern), "stock_fin_mois"].sum()
250    fin = pd.to_datetime(str(int(dern)), format="%Y%m%d")
251    v = mvt[mvt["type_mouvement"].eq("vente") & mvt["date"].between(fin - pd.Timedelta(days=89), fin)]
252    ventes_jour = v["quantite"].abs().sum() / 90
253    return _div(stock_qte, ventes_jour)

256def _rotation(d):
257    stk, mvt = d["stk"], d["mvt"]
258    v = mvt[mvt["type_mouvement"].eq("vente")]
259    if v.empty or stk.empty:
260        return np.nan
261    jours = (v["date"].max() - v["date"].min()).days + 1
262    cout_ventes_an = v["valeur_ht"].abs().sum() / jours * 365
263    stock_moyen = stk.groupby("date_sk")["valeur_stock_ht"].sum().mean()
264    return _div(cout_ventes_an, stock_moyen)

578def calculer_kpi(p: dict, kpis: list[KPI] = CATALOGUE, details: bool = True) -> pd.DataFrame:
579    """Calcule chaque KPI en global, par année, et (si details) sur ses axes de détail.
580    Renvoie un tableau long : code, domaine, kpi, unite, axe, modalite, valeur."""
581    lignes = []
582    for k in kpis:
583        d = {t: p[t] for t in k.tables}
584        lignes.append((k.code, k.domaine, k.nom, k.unite, "global", "Total", k.calcul(d)))
585        axes = ([] if k.code in KPI_SANS_ANNEE else ["annee"]) + (k.axes if details else [])
586        for ax in axes:
587            base = next((p[t] for t in k.tables if ax in p[t].columns), None)
588            if base is None:
589                continue
590            for val in sorted(base[ax].dropna().unique(), key=str):
591                v = k.calcul(_filtrer(p, k.tables, ax, val))
592                mod = str(int(val)) if ax == "annee" else str(val)
593                lignes.append((k.code, k.domaine, k.nom, k.unite, ax, mod, v))
594    res = pd.DataFrame(lignes, columns=["code", "domaine", "kpi", "unite", "axe", "modalite", "valeur"])
595    res["valeur"] = pd.to_numeric(res["valeur"], errors="coerce")
596    return res

707def controler_coherence(p: dict, controles: list[Controle] = CONTROLES) -> pd.DataFrame:
708    lignes = []
709    for c in controles:
710        try:
711            v = float(c.calcul(p))
712        except Exception as e:                      # un contrôle ne doit pas bloquer les autres
713            lignes.append((c.code, c.description, np.nan, c.attendu, c.tolerance, c.unite, f"ERREUR : {e}"))
714            continue
715        ok = abs(v - c.attendu) <= c.tolerance + 1e-12
716        lignes.append((c.code, c.description, v, c.attendu, c.tolerance, c.unite, "CONFORME" if ok else "ÉCART"))
717    return pd.DataFrame(lignes, columns=["code", "controle", "valeur", "attendu", "tolerance", "unite", "resultat"])

720def recette_dwh(res: pd.DataFrame, chemin_attendus: str, seuil_relatif: float = 0.001) -> pd.DataFrame:
721    """Recette niveau 1 : KPI calculés sur le DWH  ↔  mêmes KPI calculés sur donnees_reference.
722    OK si l'écart relatif est ≤ seuil (0,1 % par défaut) ; sinon « À EXPLIQUER »."""
723    att = pd.read_csv(chemin_attendus, dtype={"modalite": str}, encoding="utf-8-sig")
724    r = res.merge(att[["code", "axe", "modalite", "valeur_attendue"]], on=["code", "axe", "modalite"], how="outer")
725    r["kpi"] = r["kpi"].fillna(r["code"].map(lambda c: KPI_PAR_CODE[c].nom if c in KPI_PAR_CODE else c))
726    r["ecart"] = r["valeur"] - r["valeur_attendue"]
727    r["ecart_relatif"] = np.where(r["valeur_attendue"].abs() > 0, r["ecart"] / r["valeur_attendue"].abs(),
728                                  np.where(r["ecart"].abs() > 0, np.inf, 0))
729
730    def statut(x):
731        if pd.isna(x["valeur"]) and pd.isna(x["valeur_attendue"]):
732            return "OK"
733        if pd.isna(x["valeur"]) or pd.isna(x["valeur_attendue"]):
734            return "ABSENT"
735        return "OK" if abs(x["ecart_relatif"]) <= seuil_relatif else "À EXPLIQUER"
736
737    r["statut"] = r.apply(statut, axis=1)
738    ordre = {k.code: i for i, k in enumerate(CATALOGUE)}
739    r["_o"] = r["code"].map(ordre)
740    r["_a"] = r["axe"].map({"global": 0, "annee": 1}).fillna(2)
741    r = r.sort_values(["_o", "_a", "axe", "modalite"], kind="stable")
742    return r[["code", "kpi", "axe", "modalite", "valeur", "valeur_attendue", "ecart", "ecart_relatif", "statut"]]
